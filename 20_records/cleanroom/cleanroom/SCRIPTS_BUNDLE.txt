\\ CLEAN-ROOM shared census library  (v3)

\\ Newton polygon slope-0 (unit-root) count for monic pc in ZZ[x], at p=3:
\\ count of roots with v3 = 0 equals n - i0, i0 = least i with v3(coeff_i) = 0.
np_ord_count(pc) =
{
  my(n = poldegree(pc), i0 = -1);
  for(i = 0, n,
    my(c = polcoef(pc, i));
    if(c != 0 && valuation(c, 3) == 0, i0 = i; break));
  if(i0 < 0, return(0));
  n - i0;
}

\\ Stable lattice, upward span, Howell forms mod 3^K.
\\ Ms = vector of exact rational matrices. Returns [B, t, rounds]; lattice = span(B)/3^t.
build_lattice(Ms, K) =
{
  my(n = matsize(Ms[1])[1], D = denominator(Ms[1]), m3 = 3^K);
  for(i = 2, #Ms, D = lcm(D, denominator(Ms[i])));
  my(v3D = valuation(D, 3), u = D / 3^valuation(D,3));
  my(uinv = Mod(u, m3)^(-1));
  my(Nk = vector(#Ms, i, Mod(D*Ms[i], m3) * uinv));  \\ = Mod(3^v3D * M_i, m3)
  my(B = matid(n), t = 0, rounds = 0, sc = 3^v3D);
  while(rounds < 12,
    rounds++;
    my(Bm = Mod(B, m3));
    my(gens = vector(1 + #Ms));
    gens[1] = sc * B;
    for(i = 1, #Ms, gens[1 + i] = lift(Nk[i] * Bm));
    my(H = matimagemod(matconcat(gens), m3));
    my(Hstab = matimagemod(sc * B, m3));
    if(H == Hstab, return([B, t, rounds]));
    B = H; t += v3D;
  );
  error("lattice did not stabilize in 12 rounds");
}

\\ Joint primary block decomposition over F3 of commuting matrices Xbs (intmod-3 entries).
blocksplit(Xbs) =
{
  my(blocks = List([matid(matsize(Xbs[1])[1]) * Mod(1,3)]));
  for(o = 1, #Xbs,
    my(nb = List());
    for(b = 1, #blocks,
      my(W = blocks[b]);
      my(R = matinverseimage(W, Xbs[o] * W));
      my(f = factormod(lift(charpoly(R)), 3));
      if(matsize(f)[1] == 1,
        listput(nb, W),
        for(r = 1, matsize(f)[1],
          my(q = f[r,1], m = f[r,2]);
          my(Q = subst(q, variable(q), R)^m);
          my(Kq = matker(Q));
          listput(nb, W * Kq));
      );
    );
    blocks = nb;
  );
  blocks;
}

\\ per-block packet table: for block basis W, ops Xbs -> [dim, [q,m] per op]  (q lifted to ZZ[x])
blockdata(Xbs, W) =
{
  my(row = vector(1 + #Xbs));
  row[1] = matsize(W)[2];
  for(o = 1, #Xbs,
    my(R = matinverseimage(W, Xbs[o] * W));
    my(f = factormod(lift(charpoly(R)), 3));
    if(matsize(f)[1] != 1, error("block not primary for op ", o));
    row[1 + o] = [lift(f[1,1]), f[1,2]];
  );
  row;
}

\\ fixed-width canonical sort key for a block row (pads coeff vectors to length 12)
blockkey(row) =
{
  my(v = [row[1]]);
  for(o = 2, #row,
    my(q = row[o][1], cv = Vec(q));
    cv = concat(cv, vector(12 - #cv, z, 0));
    v = concat(v, concat([poldegree(q)], cv));
    v = concat(v, [row[o][2]]));
  v;
}

\\ max |entry| of an integer matrix
maxent(B) =
{
  my(me = 0, n = matsize(B)[2]);
  for(j = 1, n, me = max(me, vecmax(abs(Vec(B[,j])))));
  me;
}

\\ dual polynomial under alpha -> c/alpha (roots map), monic over F3
qdual(q, c) =
{
  my(d = poldegree(q), a0 = polcoef(q,0), r = 0);
  for(i = 0, d, r += polcoef(q, i) * c^i * 'x^(d-i));
  r = r * Mod(1,3); r = lift(r / polcoef(r, d));
  r;
}

\\ intra-packet Frobenius pairing: exists root alpha of q (deg d>=2) and k in 1..d-1
\\ with alpha^(1+3^k) = c ? return list of such k
frobpair(q, c) =
{
  my(d = poldegree(q));
  if(d < 2, return([]));
  my(g = ffgen([3, d], 'w));
  my(rts = polrootsmod(q * (g^0), g));
  if(#rts == 0, return([]));
  my(al = rts[1], res = List());
  for(k = 1, d-1,
    if(al^(1 + 3^k) == c * g^0, listput(res, k)));
  Vec(res);
}

def loc_icosa(classes, ramfac, swap=False, scramble=0.0, classcram=0.0, rng=None):
    def f(p):
        if p in ramfac: return ramfac[p]
        c = classes.get(p)
        if c is None: return np.array([1.0])
        c = int(c)
        if c < 0: return np.array([1.0])
        if rng is not None and c in (3,4) and rng.random() < scramble:
            c = 7 - c
        if rng is not None and c in (1,2) and rng.random() < classcram:
            c = 3 - c
        if swap and c in (3,4): c = 7 - c
        if c == 0: return np.array([1.0,-3.0,3.0,-1.0])                       # (1-T)^3
        if c == 1: return np.array([1.0,1.0,-1.0,-1.0])                       # (1-T)(1+T)^2
        if c == 2: return np.array([1.0,0.0,0.0,-1.0])                        # 1-T^3
        if c == 3: return np.convolve([1.0,-1.0],[1.0,-G1,1.0])              # (1-T)(1-g1 T+T^2)
        if c == 4: return np.convolve([1.0,-1.0],[1.0, G2,1.0])              # (1-T)(1+g2 T+T^2)
        return np.array([1.0])
    return f

#!/usr/bin/env python3
"""Rung E PoC stage 4: the G_new port meter, construction-level.
Two arms A(u)=e^{u/2}F(e^u), A(-u); x(u)=arm imbalance; G_new=2/(1-x); eta=artanh x.
Keyed pole control (zeta*L5*L8, pole at s=1 => delta=1/2) calibrates the jet-ratio pole locator."""
import numpy as np, mpmath as mp, json
mp.mp.dps = 22
from scipy.interpolate import CubicSpline

lg, vals = np.load("kernel_grid.npy")
SP = CubicSpline(lg, vals)
lo, hi = float(np.exp(lg[0])), float(np.exp(lg[-1]))
def phi3(y):
    y = np.asarray(y, dtype=float); out = np.zeros_like(y)
    m = y < hi; yy = np.clip(y[m], lo, hi); out[m] = SP(np.log(yy)); return out
PI32 = float(mp.pi)**1.5
G1 = (np.sqrt(5)-1)/2; G2 = (np.sqrt(5)+1)/2

def sieve(X, locfac):
    a = np.zeros(X+1); a[1] = 1.0
    isc = np.ones(X+1, dtype=bool); isc[:2] = False
    for i in range(2, int(X**0.5)+1):
        if isc[i]: isc[i*i::i] = False
    for p in np.nonzero(isc)[0]:
        p = int(p); kmax = int(np.log(X)/np.log(p))
        poly = locfac(p)
        inv = np.zeros(kmax+1); inv[0] = 1.0
        for k in range(1, kmax+1):
            s = 0.0
            for j in range(1, min(k, len(poly)-1)+1): s += poly[j]*inv[k-j]
            inv[k] = -s
        b = a.copy()
        for k in range(1, kmax+1):
            if inv[k] == 0.0: continue
            pk = p**k
            if pk > X: break
            b[pk::pk] += inv[k]*a[1:(X//pk)+1]
        a = b
    return a

def kron(D, p):
    if p == 2:
        return {1:1, 7:1, 3:-1, 5:-1}.get(D % 8, 0)
    if D % p == 0: return 0
    return 1 if pow(D % p, (p-1)//2, p) == 1 else -1

def load_classes(fn):
    d = {}
    for line in open(fn):
        p, c = line.strip().split(":"); d[int(p)] = c
    return d

def loc_icosa(classes, ram, swap=False):
    def f(p):
        if p in ram: return ram[p]
        c = classes.get(p)
        if c is None: return np.array([1.0])
        c = int(c)
        if c < 0: return np.array([1.0])
        if swap and c in (3,4): c = 7 - c
        return {0: np.array([1.,-3.,3.,-1.]), 1: np.array([1.,1.,-1.,-1.]),
                2: np.array([1.,0.,0.,-1.]),
                3: np.convolve([1.,-1.],[1.,-G1,1.]), 4: np.convolve([1.,-1.],[1.,G2,1.])}[c]
    return f

def meter(a, X, N, ugrid):
    """returns x(u), eta(u), G_new(u) from the two theta arms."""
    n = np.arange(1, X+1)
    xs, es, gs = [], [], []
    for u in ugrid:
        t = float(np.exp(u))
        A  = float(np.exp(u/2) * np.dot(a[1:], phi3(n*(PI32*t/np.sqrt(N)))))
        Am = float(np.exp(-u/2) * np.dot(a[1:], phi3(n*(PI32/t/np.sqrt(N)))))
        x = (Am - A)/(Am + A)          # reflection coefficient at the port
        xs.append(x); es.append(np.arctanh(np.clip(x, -0.999999, 0.999999))); gs.append(2.0/(1.0-x))
    return np.array(xs), np.array(es), np.array(gs)

def jetfit(ugrid, eta):
    """fit eta(u) = c1 u + c3 u^3 (odd jets); return c1, c3, delta_hat = sqrt(6 c3/c1) if sane."""
    Uf = np.vstack([ugrid, ugrid**3]).T
    c, *_ = np.linalg.lstsq(Uf, eta, rcond=None)
    c1, c3 = c
    dh = float(np.sqrt(6*c3/c1)) if c1 != 0 and c3/c1 > 0 else float('nan')
    return float(c1), float(c3), dh

UG = np.array([0.03, 0.06, 0.09, 0.13, 0.17, 0.21, 0.25])

def report(name, a, X, N):
    xs, es, gs = meter(a, X, N, UG)
    c1, c3, dh = jetfit(UG, es)
    print(f"{name}:")
    print(f"  G_new(u): " + " ".join(f"{g:.10f}" for g in gs[:4]) + " ...")
    print(f"  max|G_new - 2| = {np.max(np.abs(gs-2)):.3e}   eta-jets c1={c1:.4e} c3={c3:.4e}   delta_hat={dh:.4f}")
    return {"name": name, "maxdev": float(np.max(np.abs(gs-2))), "c1": c1, "c3": c3, "delta_hat": dh}
